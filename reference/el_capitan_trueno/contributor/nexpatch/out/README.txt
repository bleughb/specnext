(the .nex files first: cd trueno/build; python make_nex.py; python make_nex.py --en; python make_nex.py --immortal; python make_nex.py --en --immortal)
python nexpatch/nexpatch.py make "trueno/Capitan Trueno, El (1990)(Dinamic)(ES).z80" -o "nexpatch/out/Capitan Trueno.html" --choice "Language" --variant "Español" "" trueno/Trueno_sprites.nex --variant "English" " (Eng)" trueno/Trueno_sprites_en.nex --option "Immortality (the energy never drops)" " (Immortal)" trueno/Trueno_sprites.nex trueno/Trueno_sprites_immortal.nex --title "El Capitan Trueno" --subtitle "no clash patcher for the ZX Spectrum Next" --about "El Capitan Trueno (Dinamic, 1990), part 1, without colour clash. The game itself is untouched and plays exactly like the original. The colour on/off item has been removed from the title menu: it no longer makes sense here. You can choose the original Spanish texts or an English translation, and switch on immortality if you just want to look around." --notice "This page stores no game resources: only the changes, which can be read only together with the original snapshot.
