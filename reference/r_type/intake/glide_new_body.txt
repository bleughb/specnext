; ---- before spr_list is published ------------------------------------------------------------
g_publish:
        push iy
        xor a
        ld (g_mov_n),a
        ld (g_k),a
        ld a,(g_fr)             ; P = frames since the last publish
        ld c,a
        xor a
        ld (g_fr),a
        ld a,c
        cp 2
        jp c,.done              ; one frame or none: nothing to spread
        cp 9
        jr c,.pk
        ld a,8
.pk:    ld (g_p),a
        dec a
        ld (g_kn),a
        ld a,(g_p)
        ld hl,g_recip
        add hl,a
        ld a,(hl)
        ld (g_r),a
        ld a,(g_cur_n)
        or a
        jp z,.done
        ld ix,g_cur
        ld b,a
.e:     push bc
        ld a,(ix+4)
        or a
        jp z,.nx                ; no sprites (drawn on the ULA)
        ld a,(g_prev_n)
        or a
        jp z,.nx
        ld b,a
        ld hl,g_prev
.f:     ld a,(hl)
        cp (ix+0)
        jr nz,.fn
        inc hl
        ld a,(hl)
        dec hl
        cp (ix+1)
        jr nz,.fn
        inc hl
        inc hl
        ld a,(hl)
        dec hl
        dec hl
        cp (ix+2)
        jr z,.found
.fn:    ld a,FENT
        add hl,a
        djnz .f
        jp .nx
.found: push hl
        pop iy                  ; IY = the same figure last update
        ld a,(ix+5)
        sub (iy+5)
        ld hl,g_ax
        call axis
        jp c,.nx
        ld a,(ix+6)
        sub (iy+6)
        ld hl,g_ay
        call axis
        jp c,.nx
        ld hl,(g_ax+2)
        ld a,h
        or l
        ld hl,(g_ay+2)
        or h
        or l
        jp z,.nx                ; did not move
        ld a,(ix+3)             ; each of its sprites
        ld c,a
        ld b,(ix+4)
.sp:    push bc
        ld a,(g_mov_n)
        cp MAXSPR
        jp nc,.spx
        ld l,c                  ; HL = spr_list + 5 * sprite
        ld h,0
        ld e,l
        ld d,h
        add hl,hl
        add hl,hl
        add hl,de
        ld de,spr_list
        add hl,de
        push hl
        ld a,(g_mov_n)          ; IY = mov record
        ld e,a
        ld d,MENT
        mul d,e
        ld iy,g_mov
        add iy,de
        pop hl
        ld (iy+0),c
        ld a,(hl)               ; target x (9 bits), y, attr 2
        ld (iy+1),a
        inc hl
        ld a,(hl)
        ld (iy+3),a
        inc hl
        ld a,(hl)
        and 1
        ld (iy+2),a
        ld a,(hl)
        and 0xFE
        ld (iy+4),a
        dec hl
        dec hl
        ld de,(g_ax)            ; offsets and steps
        ld (iy+5),e
        ld (iy+6),d
        ld de,(g_ay)
        ld (iy+7),e
        ld (iy+8),d
        ld de,(g_ax+2)
        ld (iy+9),e
        ld (iy+10),d
        ld de,(g_ay+2)
        ld (iy+11),e
        ld (iy+12),d
        call g_pos              ; shown position now -> spr_list
        ld (hl),e
        inc hl
        ld (hl),c
        inc hl
        ld a,(iy+4)
        or d
        ld (hl),a
        ld hl,g_mov_n
        inc (hl)
.spx:   pop bc
        inc c
        dec b
        jp nz,.sp
.nx:    ld de,FENT
        add ix,de
        pop bc
        dec b
        jp nz,.e
        ld a,(g_kn)
        ld (g_k),a
.done:  ld hl,g_cur             ; this update becomes the previous one
        ld de,g_prev
        ld bc,MAXF*FENT
        ldir
        ld a,(g_cur_n)
        ld (g_prev_n),a
        xor a
        ld (g_cur_n),a
        pop iy
        ret

; IY = mov record -> E = x low, D = x bit 8, C = y: target + rounded offset.
g_pos:  ld e,(iy+5)
        ld d,(iy+6)
        call .rnd               ; A = whole pixels (signed)
        ld e,a
        rlca
        sbc a,a
        ld d,a
        push hl
        ld l,(iy+1)
        ld h,(iy+2)
        add hl,de
        ld e,l
        ld a,h
        and 1
        ld d,a
        ld l,(iy+7)
        ld h,(iy+8)
        ex de,hl
        push hl
        call .rnd
        pop hl
        ex de,hl
        add a,(iy+3)
        ld c,a
        pop hl
        ret
.rnd:   ex de,hl                ; DE = 8.8 -> A = round(DE / 256)
        ld de,0x80
        add hl,de
        ld a,h
        ret

; A = signed move in cells, HL = 4 bytes: offset now (8.8), step per frame (8.8).
; Carry if more than 4 cells (a jump, not a move).
axis:   push hl
        ld (hl),0
        inc hl
        ld (hl),0
        inc hl
        ld (hl),0
        inc hl
        ld (hl),0
        pop hl
        or a
        ret z
        ld c,0
        bit 7,a
        jr z,.p
        neg
        dec c                   ; C = FF: negative
.p:     cp 5
        ccf
        ret c
        add a,a
        add a,a
        add a,a                 ; pixels
        push hl
        ld d,a
        ld a,(g_r)
        ld e,a
        mul d,e                 ; DE = step (8.8) = pixels * 256 / P
        ld a,d                  ; (pixels in A was moved to D; keep for the offset)
        pop hl
        push de
        ld b,a
        ; offset = step - pixels*256  (negative of the remaining distance)
        ex de,hl                ; HL = step
        push de
        ld de,0
        ld d,b
        ld b,0
        pop de
        pop de
        ret
